SELECT DISTINCT
    CONCAT(STPO_B2.IDNRK, MAST_B2.MATNR) AS ConcatenacaoColunas,
    STPO_B2.IDNRK AS COMPONENTE,
    MAKT_PS0.MAKTX AS DESCRICAO_COMPP,
    STPO_B2.STLNR AS LT,
    STPO_B2.SANFE AS RELEVANCIA_PROD,
    STPO_B2.SANKA AS RELEVANCIA_CUSTO,
    STPO_B2.AEDAT AS MODIFICACAO,
    STPO_B2.MENGE AS QUANTIDADE,
    MAST_B2.MATNR AS PN_FINAL,
    MAST_B2.WERKS AS CENTRO,
    MARA_B2.MSTAE AS STATUS_LT,
    MKAL_B2.MDV01 AS LINHA_AA,
    MARC_B2.SCHGT AS FLAG_GRANEL
FROM MARD_DALI.STPO_B2 STPO_B2
INNER JOIN MARD_PSLA.MAST_B2 ON STPO_B2.STLNR = MAST_B2.STLNR
INNER JOIN MARD_PSLA.MARA_B2 ON MAST_B2.MATNR = MARA_B2.MATNR
INNER JOIN MARD_PSLA.MKAL_B2 ON MAST_B2.MATNR = MKAL_B2.MATNR
INNER JOIN MARD_PSLA.MARC_B2 ON MAST_B2.MATNR = MARC_B2.MATNR
INNER JOIN MARD_DALI_BBM.MAKT_PS0 ON MAKT_PS0.MATNR = STPO_B2.IDNRK
WHERE MAST_B2.WERKS = '6854'
  AND MKAL_B2.MDV01 = '5303-001'
  AND MAKT_PS0.SPRAS = 'P'
  AND MAST_B2.STLAN = 3
AND TO_DATE(STPO_B2.DATUV, 'YYYYMMDD') = (
  SELECT MAX(TO_DATE(STPO_SUB.DATUV, 'YYYYMMDD'))
  FROM MARD_PSLA.STPO_B2 STPO_SUB
  WHERE STPO_SUB.STLNR = STPO_B2.STLNR
    AND STPO_SUB.POSNR = STPO_B2.POSNR
    AND TO_DATE(STPO_SUB.DATUV, 'YYYYMMDD') <= CURRENT_DATE
)
